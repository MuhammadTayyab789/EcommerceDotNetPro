        public Register(RequestData request)
        {
            returnresponse = new ExchangeResponse<cRequestAlfaRegistrationViaRapid>();
            reqlog.FunctionName = "AlfaRegistrationViaRapid";
            cDecription<cRequestAlfaRegistrationViaRapid> exchangeSession = new();

            try
            {
                returnresponse = exchangeSession.ExchangeSession(request.requestAlfaReg, request.requestAlfaReg.SessionID, request.IMEI, "", "", "", request.Hash);
                request.requestAlfaReg = returnresponse.Data;
            }
            catch (Exception ex)
            {
                reqlog.LogType = AlfaService.Configuration.cEnum.LogType.Error;
                returnresponse.ResponseCode = "01";
                returnresponse.ResponseDesc = AppResponseCodes.AppErrorCodes[returnresponse.ResponseCode].ToString();

                logstr.Add("Exception while Exchanging Session with AlfaRegistrationViaRapid " + ex);
            }
        }

        public cResponseData FuncAlfaRegistrationViaRapid(RequestData requestData)
        {
            reqlog.FunctionName = "AlfaRegistrationViaRapid";

            AlfaRegHelper alfaRegHelper = new();
            List<RapidAccountOpeningDetailsWithoutDocuments> rapidCustomers = new();
            PersonalDetails personalDetails = new();
            BankingDetails bankingDetails = new();

            cResponseData response = new()
            {
                responseAlfaReg = new cResponseAlfaRegistrationViaRapid(),
            };

            cEncription<cResponseAlfaRegistrationViaRapid> encryption = new();
            try
            {
                if (returnresponse.ResponseCode.Equals("00"))
                {
                    reqlog.FunctionName = " AlfaRegistrationViaRapid | CNIC : " + requestData.requestAlfaReg.CNIC;
                   
                    logstr.Add(" request " + JsonConvert.SerializeObject(requestData.requestAlfaReg));


                    if (!string.IsNullOrEmpty(requestData.requestAlfaReg.CNIC) && !string.IsNullOrEmpty(requestData.requestAlfaReg.LoginName) && !string.IsNullOrEmpty(requestData.requestAlfaReg.PIN) &&
                        !string.IsNullOrEmpty(requestData.requestAlfaReg.MobileNo) && !string.IsNullOrEmpty(requestData.requestAlfaReg.Password) && !string.IsNullOrEmpty(requestData.requestAlfaReg.AccountNumber))
                    {
                        rapidCustomers = alfaRegHelper.GetCustomerData(requestData.requestAlfaReg.CNIC, requestData.IMEI, ref logstr);
                        logstr.Add(" RapidData :" + JsonConvert.SerializeObject(rapidCustomers));

                        if (rapidCustomers != null && rapidCustomers.Count > 0)
                        {

                            personalDetails = JsonConvert.DeserializeObject<PersonalDetails>(rapidCustomers[0].PersonalDetails);
                            bankingDetails = JsonConvert.DeserializeObject<BankingDetails>(rapidCustomers[0].BankingDetails);

                            bool IsCustomerExist = alfaRegHelper.IsCustomerExists(requestData.requestAlfaReg.CNIC, requestData.requestAlfaReg.LoginName, ref logstr);

                            bool IsAccountExist = alfaRegHelper.IsAccountExists(requestData.requestAlfaReg.CNIC, requestData.requestAlfaReg.LoginName, ref logstr);

                            bool IsUserNameExist = alfaRegHelper.IsUserNameExists(requestData.requestAlfaReg.LoginName, ref logstr);


                            if (rapidCustomers[0].IsBioVerified.Equals('0'))
                            {

                                response.ResponseCode = "950";
                                response.ResponseDesc = AppResponseCodes.AppErrorCodes[response.ResponseCode].ToString();

                            }

                            if (IsCustomerExist)
                            {
                                response.ResponseCode = "951";
                                response.ResponseDesc = AppResponseCodes.AppErrorCodes[response.ResponseCode].ToString();


                            }
                            if (IsAccountExist)
                            {
                                response.ResponseCode = "952";
                                response.ResponseDesc = AppResponseCodes.AppErrorCodes[response.ResponseCode].ToString();

                            }
                            if (IsUserNameExist)
                            {
                                response.ResponseCode = "956";
                                response.ResponseDesc = AppResponseCodes.AppErrorCodes[response.ResponseCode].ToString();

                            }




                            string MobileNumber = alfaRegHelper.FormatMobileNumber(requestData.requestAlfaReg.MobileNo, ref logstr);
                            logstr.Add("MobileNumber: " + MobileNumber);

                            string PIN = alfaRegHelper.EncryptPIN(requestData.requestAlfaReg.PIN, ref logstr);

                            logstr.Add("PIN: " + PIN);

  

                            if (!string.IsNullOrEmpty(PIN))
                            {
                                response.ResponseCode = "953";
                                response.ResponseDesc = AppResponseCodes.AppErrorCodes[response.ResponseCode].ToString();

                            }
                           

                            string Password = alfaRegHelper.EncryptPassword(requestData.requestAlfaReg.Password, ref logstr);

                            logstr.Add("Password: " + Password);

                            if (string.IsNullOrEmpty(requestData.requestAlfaReg.CategoryName))
                            {

                                alfaRegHelper.InsertUser(requestData.requestAlfaReg.CNIC, requestData.requestAlfaReg.MobileNo, requestData.requestAlfaReg.MobileNo, requestData.requestAlfaReg.AccountNumber, PIN, Password, ref logstr);

                                response.ResponseCode = "00";
                                response.ResponseDesc = AppResponseCodes.AppErrorCodes[response.ResponseCode].ToString();


                            }




                            
                            

                                if (!string.IsNullOrEmpty(MobileNumber) && (!string.IsNullOrEmpty(PIN)) && (!string.IsNullOrEmpty(Password)))
                                {

                                    bool result = alfaRegHelper.CreateAlfaUser(alfaRegHelper.PrepareRegistrationRequest(requestData.requestAlfaReg.CNIC, MobileNumber, MobileNumber, rapidCustomers[0].Email,
                                                        Password, requestData.requestAlfaReg.LoginName, requestData.requestAlfaReg.AccountNumber,
                                                         "", "", (int)General.UserStatus.InActive, "", 5, personalDetails.DateOfBirth.ToString(), _ServiceFunctions.GetMVisaPan(),
                                                         personalDetails.Name, "0", returnresponse.DeviceID, returnresponse.Device_OS, returnresponse.AppVersion, _ServiceFunctions.GetIPAddress(),
                                   "Digital Bio", "", "", "Alfa", "", ref logstr, ref reqlog), ref logstr, ref reqlog);


                                }



                            
                            catch (Exception ex)
                            {

                                logstr.Add("Exception while creating user " + ex);


                            }

                        }
                        else
                        {

                            logstr.Add("User Data not exist");
                            response.ResponseCode = "534";        
                            response.ResponseDesc = AppResponseCodes.AppErrorCodes["534"].ToString();

                        }
                    }
                    else
                    {

                        response.ResponseCode = "10";
                        response.ResponseDesc = AppResponseCodes.AppErrorCodes[response.ResponseCode].ToString();


                    }


                }
                else
                {
                    logstr.Add("Return response  : " + returnresponse.ResponseCode);
                    response.ResponseCode = returnresponse.ResponseCode;
                    response.ResponseDesc = AppResponseCodes.AppErrorCodes[response.ResponseCode].ToString();

                }
            }
            catch (Exception ex)
            {
                reqlog.LogType = AlfaService.Configuration.cEnum.LogType.Error;
                logstr.Add(" AlfaRegistrationViaRapid Exception : " + ex);
                response.ResponseCode = "555";
                response.ResponseDesc = AppResponseCodes.AppErrorCodes[response.ResponseCode].ToString();

            }
            finally
            {
                SQMService sQMService = new();
                #region :: Encrypting Details
                response.responseAlfaReg.SessionID = returnresponse.SessionID;
                ServiceFunctions _ServiceFunctions = new();
                response.responseAlfaReg.AppURL = _ServiceFunctions.GetSQMUrl(sQMService, requestData.IMEI, returnresponse.LoginName, reqlog.FunctionName, returnresponse.SessionID, returnresponse.OldSessionID, ref logstr);
                response.responseAlfaReg.AppVersion = returnresponse.AppVersion;
                response.responseAlfaReg = encryption.FuncEncripto(response.responseAlfaReg, returnresponse.OldSessionID);
                #endregion
                reqlog.logger = logstr;
                _logger.infos(reqlog);
            }
            return response;
        }






    }

        public string FormatMobileNumber(string number, ref List<string> logstr)
        {


            if (number.StartsWith("+92"))
            {
                return number;
            }
            if (number.StartsWith("92"))
            {
                return "+" + number;
            }
            else if (number.StartsWith("0"))
            {

                return "+92" + number.Substring(1);
            }

            else
            {
                return number;
            }


        }

        public string ValidateAndEncryptPin(string PIN, ref List<string> logstr)
        {
            string encrytpPIN = string.Empty;

            bool chkConsecutive = _ServiceFunctions.HasConsecutiveChars(PIN, 4);
            bool chkSequence = _ServiceFunctions.IsSequential(PIN);

            logstr.Add("PIN :" + PIN);
            try
            {
                if (chkConsecutive)
                {
                    logstr.Add("chkConsecutive" + chkConsecutive);
                    return string.Empty;

                }
                if (chkSequence)
                {
                    logstr.Add("chkSequence " + chkSequence);
                    return string.Empty;

                }
                else
                {

                    encrytpPIN = encryption.FuncEncryption(PIN.Trim(), cConfig.E_D_Pass);

                    logstr.Add(" encrypt pin " + encrytpPIN);
                    return encrytpPIN;

                }

            }
            catch (Exception ex)

            {
                logstr.Add("Exception in ValidateAndEncryptPin " + ex);
            }
            return encrytpPIN;


        }
